# Fix 6 Defects in c028bcb9 Implementation Plan

## Overview

Resolves the 3 critical (I1-I3) and 3 important (Q1-Q3) findings from the c028bcb9 review. Combines three layered changes: (a) gate image-promotion on `!isBudgetExceeded && !pinnedTier`; (b) drop dead `level` parameter from `getUnsupportedTiers`; (c) expand `isValidRoutingDecision` into a field-by-field sanitizer. Plan inherits 5 slices 1:1 from the design.

## Desired End State

From the design's Desired End State:

- User with `maxSessionBudget: 1.00`, cost `$1.50`, image attachment → tier `low`, `isBudgetForced: true` (Q3), image-promotion skipped (I1), notify shows `[low, budget]`, widget shows `[budget-limit]`.
- User with `pinnedTier = 'low'` + image → tier `low`, `isPinned: true`, image-promotion skipped (I2), Google guard preserved.
- User restoring a session with `lastDecision.isPinned: "yes"` (corrupt) → state accepted, `isPinned` stripped, rest preserved (Q2).

## What We're NOT Doing

Per design's Scope → Not Building:

- Q4 — `wordBoundaryRegexCache` unbounded growth (separate concern, not critical-fix scope)
- Q5 — `containsAnyWord` `\b` boundary for non-word edge chars (UX tradeoff)
- Q6 — `buildDelegationTierChain` vs `resolveAvailableTier` opposite iteration order (test-covered, intentional)
- Q7 — `extractToolCallIds` only scans `messages` (only matters for long tool-call chains)
- Q8 — `profileSupportsImages` no-registry fallback defaults to `true` (UX discussion needed)
- Q9 — Partial-stream cost recording drops falsy `usage.cost.total` (observability concern)
- A1 — One canonical label vs all applicable flags (resolved by Q1)

#### 2. `extensions/state.ts` — per-field sanitizer + simplified validator

**File**: `extensions/state.ts`
**Changes**: Replace `isValidRoutingDecision` with `sanitizeRoutingDecisionFields`. Required fields (`profile`, `tier`, `timestamp`) reject-whole; the 6 non-optional `RoutingDecision` fields (phase, targetProvider, targetModelId, targetLabel, reasoning, thinking) get defensive defaults; 11 optional fields are dropped if invalid. `isRouterPersistedState` keeps its boolean shape but only checks the 3 required fields of `lastDecision`.

```{typescript}
import type {
  RouterPinByProfile,
  RouterThinkingByProfile,
  RoutingDecision,
  RouterPersistedState,
} from './types';
import {
  isRouterTier,
  isThinkingLevel,
  isObjectRecord,
  isRouterPhase,
} from './config';

const isValidPinByProfile = (value: unknown): boolean => {
  if (!isObjectRecord(value)) return false;
  for (const tier of Object.values(value)) {
    if (tier !== undefined && !isRouterTier(tier)) return false;
  }
  return true;
};

const isValidThinkingByProfile = (value: unknown): boolean => {
  if (!isObjectRecord(value)) return false;
  for (const tierMap of Object.values(value)) {
    if (tierMap === undefined) continue;
    if (!isObjectRecord(tierMap)) return false;
    for (const level of Object.values(tierMap)) {
      if (level !== undefined && !isThinkingLevel(level)) return false;
    }
  }
  return true;
};

const isOptionalBoolean = (v: unknown): v is boolean => typeof v === 'boolean';

const isOptionalString = (v: unknown): v is string => typeof v === 'string';

const isConfidenceLevel = (
  v: unknown,
): v is NonNullable<RoutingDecision['confidence']> =>
  v === 'high' || v === 'medium' || v === 'low';

const isVetoReason = (
  v: unknown,
): v is NonNullable<RoutingDecision['vetoReason']> => v === 'explicit-hint';

const isFailureReason = (
  v: unknown,
): v is NonNullable<RoutingDecision['failureReason']> =>
  v === 'success' || v === 'unparseable';

/**
 * Per-field sanitizer for a persisted `RoutingDecision`.
 *
 * Required fields (`profile`, `tier`, `timestamp`) reject-whole — returns
 * `undefined` if any are bad. The 6 non-optional RoutingDecision fields
 * declared in `types.ts:70-79` (`phase`, `targetProvider`, `targetModelId`,
 * `targetLabel`, `reasoning`, `thinking`) are defaulted when invalid so the
 * returned value always satisfies the interface (string → `''`, phase →
 * `'implementation'`, thinking → `'medium'`). The 10 optional fields (6
 * boolean flags + 4 classifier/veto provenance) are dropped if invalid.
 */
export const sanitizeRoutingDecisionFields = (
  value: unknown,
): RoutingDecision | undefined => {
  if (!isObjectRecord(value)) return undefined;
  if (typeof value.profile !== 'string') return undefined;
  if (typeof value.tier !== 'string' || !isRouterTier(value.tier)) {
    return undefined;
  }
  if (typeof value.timestamp !== 'number') return undefined;

  // Defaults for ALL 6 non-optional RoutingDecision fields beyond the 3
  // required ones; the interface (types.ts:70-79) makes them required, so
  // we must satisfy the literal type. Invalid values get safe defaults;
  // the 10 optional fields are dropped (not set) if invalid.
  const phase = isRouterPhase(value.phase) ? value.phase : 'implementation';
  const targetProvider = isOptionalString(value.targetProvider)
    ? value.targetProvider
    : '';
  const targetModelId = isOptionalString(value.targetModelId)
    ? value.targetModelId
    : '';
  const targetLabel = isOptionalString(value.targetLabel)
    ? value.targetLabel
    : '';
  const reasoning = isOptionalString(value.reasoning) ? value.reasoning : '';
  const thinking = isThinkingLevel(value.thinking) ? value.thinking : 'medium';
  const out: RoutingDecision = {
    profile: value.profile,
    tier: value.tier,
    timestamp: value.timestamp,
    phase,
    targetProvider,
    targetModelId,
    targetLabel,
    reasoning,
    thinking,
  };
  if (isOptionalBoolean(value.isClassifier)) {
    out.isClassifier = value.isClassifier;
  }
  if (isOptionalBoolean(value.isFallback)) {
    out.isFallback = value.isFallback;
  }
  if (isOptionalBoolean(value.isBudgetForced)) {
    out.isBudgetForced = value.isBudgetForced;
  }
  if (isOptionalBoolean(value.isRuleMatched)) {
    out.isRuleMatched = value.isRuleMatched;
  }
  if (isOptionalBoolean(value.isPinned)) {
    out.isPinned = value.isPinned;
  }
  if (isOptionalBoolean(value.isVetoed)) {
    out.isVetoed = value.isVetoed;
  }
  if (isVetoReason(value.vetoReason)) {
    out.vetoReason = value.vetoReason;
  }
  if (isConfidenceLevel(value.confidence)) {
    out.confidence = value.confidence;
  }
  if (isOptionalString(value.rawText)) {
    out.rawText = value.rawText;
  }
  if (isFailureReason(value.failureReason)) {
    out.failureReason = value.failureReason;
  }
  return out;
};

/**
 * Required-field check used inside `isRouterPersistedState`.
 * Returns true iff the value is an object AND its three required fields
 * are present and well-typed. Optional fields are not checked here — the
 * consumer runs `sanitizeRoutingDecisionFields` to strip bad optional
 * fields without rejecting the whole state.
 */
const hasValidRequiredDecisionFields = (value: unknown): boolean => {
  if (!isObjectRecord(value)) return false;
  if (typeof value.profile !== 'string') return false;
  if (typeof value.tier !== 'string' || !isRouterTier(value.tier)) return false;
  if (typeof value.timestamp !== 'number') return false;
  return true;
};

export const isRouterPersistedState = (
  value: unknown,
): value is RouterPersistedState => {
  if (!isObjectRecord(value)) return false;
  const v = value;
  if (typeof v.enabled !== 'boolean') return false;
  if (typeof v.selectedProfile !== 'string') return false;
  if (typeof v.timestamp !== 'number') return false;
  if (v.accumulatedCost !== undefined) {
    if (
      typeof v.accumulatedCost !== 'number' ||
      Number.isNaN(v.accumulatedCost) ||
      !Number.isFinite(v.accumulatedCost)
    ) {
      return false;
    }
  }
  if (v.pinByProfile !== undefined && !isValidPinByProfile(v.pinByProfile))
    return false;
  if (
    v.thinkingByProfile !== undefined &&
    !isValidThinkingByProfile(v.thinkingByProfile)
  )
    return false;
  if (v.pinTier !== undefined && !isRouterTier(v.pinTier)) return false;
  if (
    v.lastDecision !== undefined &&
    !hasValidRequiredDecisionFields(v.lastDecision)
  ) {
    return false;
  }
  if (v.debugEnabled !== undefined && typeof v.debugEnabled !== 'boolean')
    return false;
  if (v.widgetEnabled !== undefined && typeof v.widgetEnabled !== 'boolean')
    return false;
  if (
    v.lastNonRouterModel !== undefined &&
    typeof v.lastNonRouterModel !== 'string'
  )
    return false;
  return true;
};

export const buildPersistedState = (
  routerEnabled: boolean,
  selectedProfile: string | undefined,
  pinnedTierByProfile: RouterPinByProfile,
  thinkingByProfile: RouterThinkingByProfile,
  debugEnabled: boolean,
  widgetEnabled: boolean,
  debugHistory: RoutingDecision[],
  lastDecision: RoutingDecision | undefined,
  lastNonRouterModel: string | undefined,
  accumulatedCost: number,
): RouterPersistedState => {
  return {
    enabled: routerEnabled,
    selectedProfile: selectedProfile ?? '',
    pinTier: selectedProfile ? pinnedTierByProfile[selectedProfile] : undefined,
    pinByProfile: { ...pinnedTierByProfile },
    thinkingByProfile: { ...thinkingByProfile },
    debugEnabled,
    widgetEnabled,
    debugHistory,
    lastDecision,
    lastNonRouterModel,
    accumulatedCost,
    timestamp: Date.now(),
  };
};
```

## Phase 2: routing.ts budget floor branch (Q3)

### Overview

Adds a third `else if` branch to the budget-downgrade chain in `decideRouting`. When the heuristic naturally lands on `low` AND budget is exceeded, the branch sets `isBudgetForced = true` with a floor-warning reasoning. Without this, the `low` tier with exceeded budget was silent (no flag, no notification).

### Changes Required

#### 1. `extensions/routing.ts` — Q3 floor branch

**File**: `extensions/routing.ts`
**Changes**: Add a third `else if` branch in the `isBudgetExceeded` chain (after the `medium → low` branch) so a `low`-tier decision with exceeded budget is flagged.

```{typescript}
let isBudgetForced = false;
if (isBudgetExceeded && tier === 'high') {
  tier = 'medium';
  phase = 'implementation';
  reasoning = `Budget exceeded. Downgraded from high to medium tier. (Original: ${reasoning})`;
  isBudgetForced = true;
} else if (isBudgetExceeded && tier === 'medium') {
  tier = 'low';
  phase = 'lightweight';
  reasoning = `Budget exceeded. Downgraded from medium to low tier. (Original: ${reasoning})`;
  isBudgetForced = true;
} else if (isBudgetExceeded && tier === 'low') {
  // Q3: floor case — cannot downgrade further; still mark budget-forced
  // so consumers (notify, widget) surface the exceeded-budget signal.
  // tier + phase stay as the heuristic picked them; reasoning is the
  // floor-warning variant (no "Downgraded" claim since no downgrade
  // happened).
  reasoning = `Budget exceeded; already at lowest tier — no further downgrade possible. (Original: ${reasoning})`;
  isBudgetForced = true;
}
```

## Phase 3: provider.ts image-promotion gates (I1 + I2)

### Overview

Gates the image-attachment tier promotion block in `provider.ts` on `!isBudgetExceeded && !pinnedTier`. Both variables are already captured as `const` snapshots at `provider.ts:299` (pinnedTier) and `provider.ts:300` (isBudgetExceeded). Without this gate, the image-promotion block silently overrides the budget cap (I1) and the pin promise (I2). The Google continuation guard's `!pinnedTier` skip at `provider.ts:344` becomes correct only when this gate is in place.

### Changes Required

#### 1. `extensions/provider.ts` — outer `if` gate on image-promotion block

**File**: `extensions/provider.ts`
**Changes**: Add `&& !isBudgetExceeded && !pinnedTier` to the outer `if (imageAttached)` condition. Block body unchanged.

```{typescript}
// I1+I2: skip promotion when budget is exceeded (the budget cap must be
// honored even when the downgraded tier lacks image-capable models) or

if (imageAttached && !isBudgetExceeded && !pinnedTier) {

      const mm = state.currentModelRegistry?.find(p, m);
      return mm?.input?.includes('image') ?? false;

      return false;
    }
  };

  const tierModels = [
    decision.targetLabel,
    ...(profile[decision.tier]?.fallbacks ?? []),
  ];
  if (!tierModels.some(checkModelSupportsImage)) {
    const tiersToTry: RouterTier[] =
      decision.tier === 'low'
        ? ['medium', 'high']
        : decision.tier === 'medium'
          ? ['high']
          : [];

    let foundTier: RouterTier | undefined;
    for (const t of tiersToTry) {
      const tModels = [
        profile[t]?.model,
        ...(profile[t]?.fallbacks ?? []),
      ].filter((m): m is string => typeof m === 'string');
      if (tModels.some(checkModelSupportsImage)) {
        foundTier = t;
        break;
      }
    }

    if (foundTier) {
      const promotedDecision = buildRoutingDecision(
        model.id,
        profile,
        foundTier,
        phaseForTier(foundTier),
        `Forced ${foundTier} tier because the originally routed ${decision.tier} tier does not support image attachments.`,
        state.thinkingByProfile[model.id],
        false,
      );
      // Carry forward provenance (modeled after routing.ts).
      promotedDecision.isPinned = decision.isPinned;
      promotedDecision.isRuleMatched = decision.isRuleMatched;
      promotedDecision.isBudgetForced = decision.isBudgetForced;
      if (decision.isClassifier) {
        promotedDecision.isClassifier = decision.isClassifier;
        promotedDecision.confidence = decision.confidence;
        promotedDecision.rawText = decision.rawText;
        promotedDecision.failureReason = decision.failureReason;
        if (decision.isVetoed) {
          promotedDecision.isVetoed = decision.isVetoed;
          promotedDecision.vetoReason = decision.vetoReason;
        }
      }
      decision = promotedDecision;
    }
  }
}
```
